## 2. 配置文件说明

### 主配置文件：`config/config.yaml`

```yaml
# ==================== 服务器配置 ====================
server:
  port: 8080                    # HTTP 监听端口
  mode: release                 # 运行模式：debug / release / test
  read_timeout: 60              # 读超时（秒）
  write_timeout: 600            # 写超时（秒）— 大文件合并需要较长超时

# ==================== MySQL 配置 ====================
mysql:
  host: 127.0.0.1               # 数据库主机
  port: 3306                    # 数据库端口
  user: root                    # 数据库用户
  password: "${MYSQL_PASSWORD}" # 数据库密码（支持环境变量）
  database: backend_db          # 数据库名
  max_open_conns: 100           # 最大打开连接数
  max_idle_conns: 10            # 最大空闲连接数
  conn_max_lifetime: 3600       # 连接最大生命周期（秒）
  charset: utf8mb4              # 字符集

# ==================== Redis 配置 ====================
redis:
  host: 127.0.0.1               # Redis 主机
  port: 6379                    # Redis 端口
  password: "${REDIS_PASSWORD}" # Redis 密码
  db: 0                         # 数据库编号
  pool_size: 100                # 连接池大小

# ==================== JWT 配置 ====================
jwt:
  secret: "${JWT_SECRET}"                    # 签名密钥（必须通过环境变量设置）
  access_token_ttl: 168h                     # Access Token 有效期：7 天
  refresh_token_ttl: 720h                    # Refresh Token 有效期：30 天
  issuer: "devkit"                           # Token 签发者

# ==================== 日志配置 ====================
log:
  level: info                   # 日志级别：debug / info / warn / error
  format: json                  # 日志格式：json / console
  output: both                  # 输出目标：stdout / file / both
  file_path: ./logs/app.log     # 日志文件路径
  max_size: 100                 # 单文件最大尺寸（MB）
  max_backups: 7                # 保留旧文件数量
  max_age: 30                   # 保留天数

# ==================== CORS 配置 ====================
cors:
  allow_origins:                # 允许的来源
    - "http://localhost:5666"
    - "http://localhost:5173"
  allow_methods:                # 允许的 HTTP 方法
    - "GET"
    - "POST"
    - "PUT"
    - "DELETE"
    - "OPTIONS"
  allow_headers:                # 允许的请求头
    - "Authorization"
    - "Content-Type"
    - "X-Device-ID"
    - "X-CSRF-Token"
    - "X-Mobile-Token-Mode"
  expose_headers:               # 暴露的响应头
    - "Content-Disposition"
  allow_credentials: true       # 允许携带凭证
  max_age: 86400                # 预检缓存时间（秒）

# ==================== 限流配置 ====================
rate_limit:
  rate: 100                     # 每秒请求数
  burst: 200                    # 突发请求数

# ==================== 存储配置 ====================
storage:
  driver: local                 # 存储驱动：local / minio / oss / cos
  local:
    path: ./uploads             # 本地存储路径
    url_prefix: /uploads        # URL 前缀
  minio:
    endpoint: ""                # MinIO 端点
    access_key: ""              # Access Key
    secret_key: ""              # Secret Key
    bucket: ""                  # 桶名
    use_ssl: false              # 是否使用 SSL
  oss:
    endpoint: ""                # OSS 端点
    access_key_id: ""           # Access Key ID
    access_key_secret: ""       # Access Key Secret
    bucket: ""                  # 桶名
    cdn_domain: ""              # CDN 域名
  cos:
    region: ""                  # COS 区域
    secret_id: ""               # Secret ID
    secret_key: ""              # Secret Key
    bucket: ""                  # 桶名
    cdn_domain: ""              # CDN 域名

# ==================== CDN 配置 ====================
cdn:
  domain: ""                    # CDN 域名
  image_style_separator: ""     # 图片样式分隔符

# ==================== ClamAV 配置 ====================
clamav:
  enabled: false                # 是否启用病毒扫描
  socket: /var/run/clamav/clamd.ctl  # ClamAV Socket 路径
  timeout: 60                   # 扫描超时（秒）

# ==================== 加密配置 ====================
crypto:
  aes_key: "${AES_KEY}"         # AES-256 密钥（32 字节，必须通过环境变量设置）

# ==================== 验证码配置 ====================
captcha:
  width: 200                    # 验证码图片宽度

  length: 5                     # 验证码字符长度
```

## 3. 环境变量配置

### 必须设置的安全环境变量

以下三个环境变量**必须**通过环境变量设置，不能使用配置文件中的占位值：

```bash
# JWT 签名密钥（推荐 64 字节随机字符串）
export JWT_SECRET="your-super-secret-jwt-key-here"

# AES-256 加密密钥（必须 32 字节）
export AES_KEY="your-32-byte-aes-key-here!!"

# 验证码加密密钥
export CAPTCHA_SECRET="your-captcha-secret-key"
```

### 生成安全密钥

```bash
# 生成 JWT_SECRET（64 字节 base64）
openssl rand -base64 64

# 生成 AES_KEY（32 字节）
openssl rand -hex 32

# 生成 CAPTCHA_SECRET
openssl rand -base64 32
```

### 默认用户密码环境变量

```bash
# 可选：设置默认用户密码（不设置则随机生成 16 位）
export DEFAULT_USER_VBEN_PASSWORD="YourSecurePassword123!"
export DEFAULT_USER_ADMIN_PASSWORD="YourSecurePassword456!"
export DEFAULT_USER_JACK_PASSWORD="YourSecurePassword789!"
```

> **注意**：如果不设置密码环境变量，系统会随机生成 16 位密码（包含大小写字母、数字、特殊字符），首次登录后必须修改密码。

### 数据库密码环境变量

```bash
export MYSQL_PASSWORD="your-mysql-password"
export REDIS_PASSWORD="your-redis-password"
```

## 5. Docker 部署

### Dockerfile

```dockerfile
# 构建阶段
FROM golang:1.22-alpine AS builder
WORKDIR /app
COPY go.mod go.sum ./
RUN go mod download
COPY . .
RUN CGO_ENABLED=0 GOOS=linux go build -o server cmd/server/main.go

# 运行阶段
FROM alpine:3.19
RUN apk --no-cache add ca-certificates tzdata
WORKDIR /app
COPY --from=builder /app/server .
COPY --from=builder /app/config ./config
COPY --from=builder /app/migrations ./migrations
EXPOSE 8080
CMD ["./server"]
```

### docker-compose.yml

```yaml
version: "3.8"

services:
  mysql:
    image: mysql:8.0
    container_name: devkit-mysql
    environment:
      MYSQL_ROOT_PASSWORD: ${MYSQL_PASSWORD}
      MYSQL_DATABASE: backend_db
    ports:
      - "3306:3306"
    volumes:
      - mysql_data:/var/lib/mysql
    command: --default-authentication-plugin=mysql_native_password
    healthcheck:
      test: ["CMD", "mysqladmin", "ping", "-h", "localhost"]
      interval: 10s
      timeout: 5s
      retries: 3

  redis:
    image: redis:7-alpine
    container_name: devkit-redis
    command: redis-server --requirepass ${REDIS_PASSWORD}
    ports:
      - "6379:6379"
    volumes:
      - redis_data:/data
    healthcheck:
      test: ["CMD", "redis-cli", "-a", "${REDIS_PASSWORD}", "ping"]
      interval: 10s
      timeout: 5s
      retries: 3

  backend:
    build: .
    container_name: devkit-backend
    ports:
      - "8080:8080"
    environment:
      - JWT_SECRET=${JWT_SECRET}
      - AES_KEY=${AES_KEY}
      - CAPTCHA_SECRET=${CAPTCHA_SECRET}
      - MYSQL_PASSWORD=${MYSQL_PASSWORD}
      - REDIS_PASSWORD=${REDIS_PASSWORD}
    volumes:
      - uploads_data:/app/uploads
      - logs_data:/app/logs
    depends_on:
      mysql:
        condition: service_healthy
      redis:
        condition: service_healthy
    restart: unless-stopped

volumes:
  mysql_data:
  redis_data:
  uploads_data:
  logs_data:
```

### 启动 Docker 环境

```bash
# 创建 .env 文件
cat > .env << EOF
JWT_SECRET=$(openssl rand -base64 64)
AES_KEY=$(openssl rand -hex 32)
CAPTCHA_SECRET=$(openssl rand -base64 32)
MYSQL_PASSWORD=your-mysql-password
REDIS_PASSWORD=your-redis-password
EOF

# 启动所有服务
make docker-up
# 或
docker-compose up -d

# 查看日志
docker-compose logs -f backend

# 停止所有服务
make docker-down
# 或
docker-compose down
```

## 6. 数据库迁移

### 自动迁移

系统启动时会自动运行 GORM AutoMigrate，创建或更新 55 个数据表：

```bash
# 单独运行迁移
make migrate
```

迁移涵盖以下模型：

| 模块 | 模型 |
|------|------|
| 用户系统 | User, Role, Menu, Group, GroupRole, UserRole, SecurityLog, LoginDevice, OAuthUser, UserPrivacy, PasswordHistory, UserRealName, RoleApplication |
| 系统配置 | SystemSetting, UploadTask, UploadedPart, FileTypeRule, RateLimitRule, ScheduledTask, Notification, NotificationRead |
| 文件系统 | FileAsset, FileFolder, FileEntry, MediaAsset, FileShare, Tag, TagRouting, FileTag, StorageBucket, StorageConfig |
| 题库系统 | ExamCategory, Exam, Subject, Category, KnowledgePoint, QuestionSource, Question, QuestionVersion, QuestionImportTask, QuestionImportItem, QuestionShare, QuestionFeedback |
| 学习系统 | UserFavorite, UserNote, PracticeRecord, WrongBook, UserCategoryBinding |
| 移动端 | Banner, QuickMenu, MyPageMenu, MobileSettings |

### 增量迁移

对于需要执行自定义 SQL 的迁移，使用 `migrations/` 目录下的脚本：

```bash
# 手动执行增量迁移
mysql -u root -p backend_db < migrations/20260612_add_notification_system.sql
```

### 完整 Schema

`migrations/schema.sql` 包含完整的数据库 DDL（41KB），可用于全新部署：

```bash
mysql -u root -p backend_db < migrations/schema.sql
```

### Nginx 配置

```nginx
server {
    listen 80;
    server_name your-domain.com;

    # 前端静态文件
    location / {
        root /path/to/frontend-admin/apps/web-antd/dist;
        try_files $uri $uri/ /index.html;

        # 缓存策略
        location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2)$ {
            expires 30d;
            add_header Cache-Control "public, immutable";
        }
    }

    # API 反向代理
    location /api/ {
        proxy_pass http://127.0.0.1:8080;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # 大文件上传支持
        client_max_body_size 500M;
        proxy_read_timeout 600s;
        proxy_send_timeout 600s;
    }

    # WebSocket 代理
    location /api/v1/ws {
        proxy_pass http://127.0.0.1:8080;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_read_timeout 86400s;
    }

    # 本地文件访问（如果使用本地存储）
    location /uploads/ {
        alias /path/to/backend-server/uploads/;
        expires 7d;
    }
}
```

```nginx
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;

    # HTTP 重定向到 HTTPS
    # ...

server {
    listen 80;
    server_name your-domain.com;
    return 301 https://$server_name$request_uri;
```

## 11. 故障排查

### 常见问题

| 问题 | 可能原因 | 解决方案 |
|------|---------|---------|
| 启动失败：密钥校验不通过 | 使用了 `changme` 占位值 | 设置真实的环境变量 |
| 数据库连接失败 | MySQL 未启动或密码错误 | 检查 MySQL 服务和配置 |
| Redis 连接失败 | Redis 未启动或密码错误 | 检查 Redis 服务和配置 |
| 401 Unauthorized | JWT 过期或黑名单 | 刷新 Token 或重新登录 |
| 403 Forbidden | 权限不足 | 检查用户角色和权限配置 |
| 429 Too Many Requests | 触发限流 | 调整限流配置或等待 |
| 文件上传失败 | 存储配置错误 | 检查存储驱动和路径配置 |

### 查看日志

```bash
# 实时查看日志
tail -f logs/app.log | jq .

# 搜索错误日志
grep '"level":"error"' logs/app.log | jq .

# 按时间范围过滤
awk '/2026-06-18T10:00/,/2026-06-18T11:00/' logs/app.log
```
