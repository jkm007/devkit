## 1. 存储接口设计

### 核心接口

```go
// pkg/storage/storage.go

// Storage 文件存储接口
type Storage interface {
    // Upload 上传文件
    Upload(ctx context.Context, objectKey string, reader io.Reader, contentType string) (string, error)

    // Download 下载文件（完整）
    Download(ctx context.Context, objectKey string) (io.ReadCloser, error)

    // DownloadRange 范围下载（支持断点续传/视频拖动）
    DownloadRange(ctx context.Context, objectKey string, offset int64, length int64) (io.ReadCloser, error)

    // Delete 删除文件
    Delete(ctx context.Context, objectKey string) error

    // GetURL 获取文件访问 URL
    GetURL(objectKey string) string

    // GetPresignedURL 获取预签名 URL（临时授权访问）
    GetPresignedURL(ctx context.Context, objectKey string, expire int64) (string, error)
}

// MultipartUploader 分片上传接口
type MultipartUploader interface {
    // InitiateUpload 初始化分片上传，返回 uploadID
    InitiateUpload(ctx context.Context, objectKey string, contentType string) (string, error)

    // UploadPart 上传单个分片，返回 ETag
    UploadPart(ctx context.Context, objectKey string, uploadID string, partNumber int, reader io.Reader, size int64) (string, error)

    // CompleteUpload 完成上传，合并分片，返回最终 URL
    CompleteUpload(ctx context.Context, objectKey string, uploadID string, parts []CompletedPart) (string, error)

    // AbortUpload 取消上传，清理临时数据
    AbortUpload(ctx context.Context, objectKey string, uploadID string) error
}
```

### 辅助类型

### 接口设计原则

| 原则 | 说明 |
|------|------|
| 最小接口 | 只暴露必要的操作方法 |
| 统一抽象 | 所有存储后端实现相同接口 |
| Context 支持 | 所有方法接受 Context，支持超时和取消 |
| 流式处理 | 使用 io.Reader/io.ReadCloser，避免大文件全量加载到内存 |

### LocalStorage

```go
// pkg/storage/local.go

type LocalStorage struct {
    cfg config.LocalConfig
}

func NewLocalStorage(cfg config.LocalConfig) (*LocalStorage, error) {
    // 确保存储目录存在
    if err := os.MkdirAll(cfg.Path, 0755); err != nil {
        return nil, err
    }
    return &LocalStorage{cfg: cfg}, nil
}

func (s *LocalStorage) Upload(ctx context.Context, objectKey string, reader io.Reader, contentType string) (string, error) {
    fullPath := s.fullPath(objectKey)

    // 创建目录
    dir := filepath.Dir(fullPath)
    if err := os.MkdirAll(dir, 0755); err != nil {
        return "", err
    }

    // 写入文件
    file, err := os.Create(fullPath)
    if err != nil {
        return "", err
    }
    defer file.Close()

    if _, err := io.Copy(file, reader); err != nil {
        return "", err
    }

    return s.GetURL(objectKey), nil
}

// 路径遍历防护
func (s *LocalStorage) validateObjectKey(objectKey string) error {
    if strings.Contains(objectKey, "..") {
        return errors.New("invalid object key: contains ..")
    }
    if filepath.IsAbs(objectKey) {
        return errors.New("invalid object key: absolute path")
    }
    return nil
}

func (s *LocalStorage) fullPath(objectKey string) string {
    return filepath.Join(s.cfg.Path, objectKey)
}

// 范围下载
func (s *LocalStorage) DownloadRange(ctx context.Context, objectKey string, offset int64, length int64) (io.ReadCloser, error) {
    fullPath := s.fullPath(objectKey)

    file, err := os.Open(fullPath)
    if err != nil {
        return nil, err
    }

    if _, err := file.Seek(offset, io.SeekStart); err != nil {
        file.Close()
        return nil, err
    }

    return &limitedReadCloser{
        ReadCloser: file,
        remaining:  length,
    }, nil
}

type limitedReadCloser struct {
    io.ReadCloser
    remaining int64
}

func (l *limitedReadCloser) Read(p []byte) (int, error) {
    if l.remaining <= 0 {
        return 0, io.EOF
    }
    if int64(len(p)) > l.remaining {
        p = p[:l.remaining]
    }
    n, err := l.ReadCloser.Read(p)
    l.remaining -= int64(n)
    return n, err
}
```

### MinIOStorage

```go
// pkg/storage/minio.go

type MinIOStorage struct {
    client   *minio.Client
    bucket   string
    endpoint string
    useSSL   bool
}

func NewMinIOStorage(cfg config.MinIOConfig) (*MinIOStorage, error) {
    // 去除协议前缀
    endpoint := strings.TrimPrefix(cfg.Endpoint, "http://")
    endpoint = strings.TrimPrefix(endpoint, "https://")

    client, err := minio.New(endpoint, &minio.Options{
        Creds:  credentials.NewStaticV4(cfg.AccessKey, cfg.SecretKey, ""),
        Secure: cfg.UseSSL,
    })
    if err != nil {
        return nil, err
    }

    // 自动创建桶
    exists, _ := client.BucketExists(context.Background(), cfg.Bucket)
    if !exists {
        client.MakeBucket(context.Background(), cfg.Bucket, minio.MakeBucketOptions{})
        // 设置为私有策略
        client.SetBucketPolicy(context.Background(), cfg.Bucket, privatePolicy)
    }

    return &MinIOStorage{
        client:   client,
        bucket:   cfg.Bucket,
        endpoint: endpoint,
        useSSL:   cfg.UseSSL,
    }, nil
}

// 分片实现：使用独立对象而非原生分片 API
func (s *MinIOStorage) UploadPart(ctx context.Context, objectKey string, uploadID string, partNumber int, reader io.Reader, size int64) (string, error) {
    partKey := fmt.Sprintf("uploads/%s/part-%06d", uploadID, partNumber)
    info, err := s.client.PutObject(ctx, s.bucket, partKey, reader, size, minio.PutObjectOptions{})
    if err != nil {
        return "", err
    }
    return info.ETag, nil
}

func (s *MinIOStorage) CompleteUpload(ctx context.Context, objectKey string, uploadID string, parts []CompletedPart) (string, error) {
    // 使用 ComposeObject 合并分片
    sources := make([]minio.SourceInfo, len(parts))
    for i, part := range parts {
        partKey := fmt.Sprintf("uploads/%s/part-%06d", uploadID, part.PartNumber)
        sources[i] = minio.SourceInfo{
            Bucket: s.bucket,
            Key:    partKey,
        }
    }

    _, err := s.client.ComposeObject(ctx, minio.CopyDestOptions{
        Bucket: s.bucket,
        Key:    objectKey,
    }, sources...)
    if err != nil {
        return "", err
    }

    // 异步清理临时分片
    go func() {
        for _, part := range parts {
            partKey := fmt.Sprintf("uploads/%s/part-%06d", uploadID, part.PartNumber)
            s.client.RemoveObject(context.Background(), s.bucket, partKey, minio.RemoveObjectOptions{})
        }
    }()

    return s.GetURL(objectKey), nil
}
```

```go
func NewCOStorage(cfg config.COSConfig) (*COStorage, error) {
    bucketURL, _ := url.Parse(fmt.Sprintf("https://%s.cos.%s.myqcloud.com", cfg.Bucket, cfg.Region))

// COS 要求去除 ETag 的引号
func (s *COStorage) CompleteUpload(ctx context.Context, objectKey string, uploadID string, parts []CompletedPart) (string, error) {
    optParts := make([]cos.Object, len(parts))
    for i, part := range parts {
        optParts[i] = cos.Object{
            PartNumber: part.PartNumber,
            // COS 的 ETag 带引号，需要去除
            ETag: strings.Trim(part.ETag, "\""),
        }
    }

    _, _, err := s.client.Object.CompleteMultipartUpload(ctx, objectKey, uploadID, &cos.CompleteMultipartUploadOptions{
        Parts: optParts,
    })
    if err != nil {
        return "", err
    }

    return s.GetURL(objectKey), nil
}
```

### 初始化流程

```go
// pkg/storage/manager.go

func InitStorage(cfg config.StorageConfig) error {
    storageMutex.Lock()
    defer storageMutex.Unlock()

    // 1. 尝试从数据库加载配置
    dbConfig, err := loadStorageConfigFromDB()
    if err == nil {
        cfg = dbConfig
    }

    // 2. 始终初始化本地存储
    localStorage, err := NewLocalStorage(cfg.Local)
    if err != nil {
        return err
    }
    storageCache["local"] = localStorage

    // 3. 确定活跃驱动
    activeDriver := determineActiveDriver(cfg)

    // 4. 创建活跃存储实例
    if activeDriver == "local" {
        currentStorage = localStorage
    } else {
        storage, err := New(cfg)
        if err != nil {
            // 外部存储初始化失败，回退到本地存储
            logger.Warn("外部存储初始化失败，回退到本地存储", zap.Error(err))
            currentStorage = localStorage
        } else {
            currentStorage = storage
            storageCache[activeDriver] = storage
        }
    }

    return nil
}
```

### 驱动选择逻辑

```go
func determineActiveDriver(cfg config.StorageConfig) string {
    // 1. 检查 sys_storage_bucket 表是否有默认桶
    defaultBucket, err := storageBucketRepo.GetDefault()
    if err == nil && defaultBucket != nil {
        return defaultBucket.Driver
    }

    // 2. 检查系统设置中的启用标志
    settings, _ := systemSettingRepo.GetByGroup("storage")

    // 优先级：COS > OSS > MinIO > Local
    if isEnabled(settings, "cos") {
        return "cos"
    }
    if isEnabled(settings, "oss") {
        return "oss"
    }
    if isEnabled(settings, "minio") {
        return "minio"
    }

    return "local"
}
```

### 热加载

```go
func RefreshStorage(cfg config.StorageConfig) error {
    // 重新从数据库加载配置
    dbConfig, err := loadStorageConfigFromDB()
    if err == nil {
        cfg = dbConfig
    }

    // 重新初始化本地存储
    localStorage, _ := NewLocalStorage(cfg.Local)

    // 确定新的活跃驱动
    activeDriver := determineActiveDriver(cfg)

    // 创建新的存储实例
    var newStorage Storage
    if activeDriver == "local" {
        newStorage = localStorage
    } else {
        newStorage, err = New(cfg)
        if err != nil {
            newStorage = localStorage
        }
    }

    // 原子替换
    storageMutex.Lock()
    currentStorage = newStorage
    storageCache["local"] = localStorage
    if activeDriver != "local" {
        storageCache[activeDriver] = newStorage
    }
    storageMutex.Unlock()

    return nil
}
```

### 获取存储实例

```go
// GetStorage 获取当前活跃存储（线程安全）
func GetStorage() Storage {
    storageMutex.RLock()
    defer storageMutex.RUnlock()
    return currentStorage
}

// GetStorageByDriver 获取指定驱动的存储（支持历史文件访问）
func GetStorageByDriver(driver string) Storage {
    storageMutex.RLock()
    if s, ok := storageCache[driver]; ok {
        storageMutex.RUnlock()
        return s
    }
    storageMutex.RUnlock()

    // 双重检查锁定：延迟初始化
    storageMutex.Lock()
    defer storageMutex.Unlock()

    if s, ok := storageCache[driver]; ok {
        return s
    }

    // 按需创建
    cfg := config.Get().Storage
    cfg.Driver = driver
    s, err := New(cfg)
    if err != nil {
        logger.Error("创建存储实例失败", zap.String("driver", driver), zap.Error(err))
        return currentStorage
    }

    storageCache[driver] = s
    return s
}
```

### 完整上传流程

```
┌──────────┐                                    ┌──────────────┐
│  Client  │                                    │ Upload Svc   │
└────┬─────┘                                    └──────┬───────┘
     │                                                 │
     │  ① POST /files/check-upload                     │
     │     { file_hash, file_name, file_size }         │

     │                                                 │
     │  ←──── { exists: true/false } ─────────────────│
```

```go
// internal/service/upload_service.go

func (s *UploadService) CheckUpload(ctx context.Context, req *CheckUploadRequest) (*CheckUploadResponse, error) {
    // 基于文件哈希查询是否已存在
    asset, err := s.fileAssetRepo.GetByHash(req.FileHash)
    if err != nil {
        // 文件不存在，需要上传
        return &CheckUploadResponse{Exists: false}, nil
    }

    // 文件已存在，增加引用计数
    s.fileAssetRepo.IncrementRefCount(asset.ID)

    return &CheckUploadResponse{
        Exists: true,
        FileID: asset.ID,
        URL:    asset.ObjectKey,
    }, nil
}
```

### 去重机制

```
┌─────────────────────────────────────────────────────────────┐
│                    文件去重流程                               │
├─────────────────────────────────────────────────────────────┤
│                                                             │
│  上传文件时：                                                │
│  1. 计算文件哈希（SHA-256）                                   │
│  2. 查询 FileAsset（WHERE file_hash = ?）                   │
│  3. 如果存在：                                               │
│     └─ 增加 RefCount，返回已有 URL                           │
│  4. 如果不存在：                                              │
│     └─ 上传文件，创建 FileAsset 记录                          │
│                                                             │
│  删除文件时：                                                │
│  1. 原子递减 RefCount（WHERE id = ? AND ref_count > 0）      │
│  2. 如果 affected rows = 0（已是 0）：                       │
│     └─ 删除物理文件                                          │
│  3. 如果 affected rows = 1（成功递减）：                      │
│     └─ 不删除物理文件（还有其他引用）                          │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

```go
// internal/repository/file_asset_repo.go

func (r *FileAssetRepository) IncrementRefCount(id uint) error {
    return r.db.Model(&model.FileAsset{}).
        Where("id = ?", id).
        Update("ref_count", gorm.Expr("ref_count + 1")).
        Error
}

func (r *FileAssetRepository) DecrementRefCountAtomic(id uint) (int64, error) {
    result := r.db.Model(&model.FileAsset{}).
        Where("id = ? AND ref_count > 0", id).
        Update("ref_count", gorm.Expr("ref_count - 1"))

    return result.RowsAffected, result.Error
}
```

```go
// pkg/storage/routing_engine.go

type RoutingEngine struct {
    rules []model.TagRouting
}

type RoutingTag struct {
    Key   string `json:"key"`
    Value string `json:"value"`
}

type RoutingResult struct {
    Driver     string `json:"driver"`
    Bucket     string `json:"bucket"`
    PathPrefix string `json:"pathPrefix"`
    RuleID     int64  `json:"ruleId"`
    RuleName   string `json:"ruleName"`
}

func NewRoutingEngine(rules []model.TagRouting) *RoutingEngine {
    // 按优先级降序排序
    sort.Slice(rules, func(i, j int) bool {
        return rules[i].Priority > rules[j].Priority
    })
    return &RoutingEngine{rules: rules}
}

func (e *RoutingEngine) Match(tags []RoutingTag) *RoutingResult {
    var defaultRule *model.TagRouting

    for _, rule := range e.rules {
        if rule.Status != 1 {
            continue
        }

        // 记录默认规则
        if rule.IsDefault {
            if defaultRule == nil {
                defaultRule = &rule
            }
            continue
        }

        // 匹配非默认规则
        if e.matchRule(rule, tags) {
            return &RoutingResult{
                Driver:     rule.Driver,
                Bucket:     rule.Bucket,
                PathPrefix: rule.PathPrefix,
                RuleID:     rule.ID,
                RuleName:   rule.RuleName,
            }
        }
    }

    // 没有匹配的规则，使用默认规则
    if defaultRule != nil {
        return &RoutingResult{
            Driver:     defaultRule.Driver,
            Bucket:     defaultRule.Bucket,
            PathPrefix: defaultRule.PathPrefix,
            RuleID:     defaultRule.ID,
            RuleName:   defaultRule.RuleName,
        }
    }

    return nil
}

func (e *RoutingEngine) matchRule(rule model.TagRouting, tags []RoutingTag) bool {
    tagMap := make(map[string]string)
    for _, t := range tags {
        tagMap[t.Key] = t.Value
    }

    switch rule.MatchType {
    case "all":
        // 所有条件都必须匹配（AND）
        for _, cond := range rule.Conditions {
            if tagMap[cond.Key] != cond.Value {
                return false
            }
        }
        return true

    case "any":
        // 至少一个条件匹配（OR）
        for _, cond := range rule.Conditions {
            if tagMap[cond.Key] == cond.Value {
                return true
            }
        }
        return false

    case "exact":
        // 标签完全一致
        if len(tagMap) != len(rule.Conditions) {
            return false
        }
        for _, cond := range rule.Conditions {
            if tagMap[cond.Key] != cond.Value {
                return false
            }
        }
        return true

    default:
        return false
    }
}
```

### 匹配模式

| 模式 | 说明 | 示例 |
|------|------|------|
| `all` | 所有条件必须匹配（AND） | file_type=image AND source=user |
| `any` | 至少一个条件匹配（OR） | file_type=image OR file_type=video |
| `exact` | 标签完全一致（数量和值都相同） | 精确匹配所有标签 |

```go
func (t *AutoTagger) LoadFromDB(rules []FileTypeRuleData) {
    t.mu.Lock()
    defer t.mu.Unlock()

    // 重置为默认值
    t.reset()

    // 覆盖数据库规则
    for _, rule := range rules {
        ext := strings.ToLower(rule.Extension)
        if !strings.HasPrefix(ext, ".") {
            ext = "." + ext
        }

        switch rule.FileType {
        case "image":
            t.imageExts[ext] = true
        case "video":
            t.videoExts[ext] = true
        case "audio":
            t.audioExts[ext] = true
        case "document":
            t.documentExts[ext] = true
        case "archive":
            t.archiveExts[ext] = true
        default:
            t.customExts[ext] = rule.FileType
        }

        t.extTypeMap[ext] = rule.FileType
    }
}
```

### 标签生成

```go
func (t *AutoTagger) GenerateTags(filename, contentType, source string) []RoutingTag {
    tags := make([]RoutingTag, 0, 4)

    // 1. 确定文件类型
    fileType := t.detectFileType(filename, contentType)
    tags = append(tags, RoutingTag{Key: "file_type", Value: fileType})

    // 2. 来源标签
    if source == "" {
        source = "user"
    }
    tags = append(tags, RoutingTag{Key: "source", Value: source})

    // 3. 敏感度标签（默认公开）
    tags = append(tags, RoutingTag{Key: "sensitivity", Value: "public"})

    return tags
}

func (t *AutoTagger) detectFileType(filename, contentType string) string {
    // 优先通过扩展名检测
    ext := strings.ToLower(filepath.Ext(filename))
    if fileType, ok := t.extTypeMap[ext]; ok {
        return fileType
    }

    // 回退到 MIME 类型前缀匹配
    if strings.HasPrefix(contentType, "image/") {
        return "image"
    }
    if strings.HasPrefix(contentType, "video/") {
        return "video"
    }
    if strings.HasPrefix(contentType, "audio/") {
        return "audio"
    }

    return "other"
}
```

```go
// pkg/storage/local.go

func (s *LocalStorage) validateObjectKey(objectKey string) error {
    // 拒绝包含 .. 的路径
    if strings.Contains(objectKey, "..") {
        return errors.New("invalid object key: contains path traversal")
    }

    // 拒绝绝对路径
    if filepath.IsAbs(objectKey) {
        return errors.New("invalid object key: absolute path not allowed")
    }

    // 拒绝空路径
    if objectKey == "" {
        return errors.New("invalid object key: empty")
    }

    return nil
}

func (s *LocalStorage) fullPath(objectKey string) string {
    return filepath.Join(s.cfg.Path, objectKey)
}

// 上传时校验
func (s *LocalStorage) Upload(ctx context.Context, objectKey string, reader io.Reader, contentType string) (string, error) {
    if err := s.validateObjectKey(objectKey); err != nil {
        return "", err
    }

    fullPath := s.fullPath(objectKey)

    // 二次校验：解析后的路径必须在基础目录下
    absBase, _ := filepath.Abs(s.cfg.Path)
    absFull, _ := filepath.Abs(fullPath)
    if !strings.HasPrefix(absFull, absBase) {
        return "", errors.New("invalid object key: path escapes base directory")
    }

    // ... 继续上传
}
```
