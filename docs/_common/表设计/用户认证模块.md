# 用户认证模块表设计

> 基于参考文档 `01-用户认证.md`，结合当前系统现状整理的待建表结构。
> 所有系统表统一使用 `sys_` 前缀。

## 一、sys_users 升级字段

在现有 `sys_users` 表基础上新增以下字段：

```sql
-- 个人信息
ALTER TABLE sys_users ADD COLUMN `nickname`   VARCHAR(50)  DEFAULT '' COMMENT '昵称' AFTER `name`;
ALTER TABLE sys_users ADD COLUMN `email`      VARCHAR(100) DEFAULT '' COMMENT '邮箱' AFTER `nickname`;
ALTER TABLE sys_users ADD COLUMN `phone`      VARCHAR(20)  DEFAULT '' COMMENT '手机号' AFTER `email`;
ALTER TABLE sys_users ADD COLUMN `avatar`     VARCHAR(255) DEFAULT '' COMMENT '头像URL' AFTER `phone`;
ALTER TABLE sys_users ADD COLUMN `gender`     TINYINT      DEFAULT 0  COMMENT '性别: 0未知 1男 2女' AFTER `avatar`;
ALTER TABLE sys_users ADD COLUMN `birthday`   DATETIME     DEFAULT NULL COMMENT '生日' AFTER `gender`;
ALTER TABLE sys_users ADD COLUMN `bio`        VARCHAR(500) DEFAULT '' COMMENT '个人简介' AFTER `birthday`;

-- 实名认证
ALTER TABLE sys_users ADD COLUMN `real_name`  VARCHAR(50)  DEFAULT '' COMMENT '真实姓名' AFTER `bio`;
ALTER TABLE sys_users ADD COLUMN `id_card`    VARCHAR(255) DEFAULT '' COMMENT '身份证号(AES加密)' AFTER `real_name`;
ALTER TABLE sys_users ADD COLUMN `is_real`    TINYINT      DEFAULT 0  COMMENT '是否已实名: 0否 1是' AFTER `id_card`;

-- 注册来源
ALTER TABLE sys_users ADD COLUMN `register_source` VARCHAR(20) DEFAULT 'web' COMMENT '注册来源: web/h5/miniapp/app' AFTER `is_real`;

-- 登录安全
ALTER TABLE sys_users ADD COLUMN `last_login_at`      DATETIME     DEFAULT NULL COMMENT '最后登录时间' AFTER `register_source`;
ALTER TABLE sys_users ADD COLUMN `last_login_ip`      VARCHAR(50)  DEFAULT '' COMMENT '最后登录IP' AFTER `last_login_at`;
ALTER TABLE sys_users ADD COLUMN `last_login_device`  VARCHAR(200) DEFAULT '' COMMENT '最后登录设备' AFTER `last_login_ip`;
ALTER TABLE sys_users ADD COLUMN `login_fail_count`   INT          DEFAULT 0  COMMENT '连续登录失败次数' AFTER `last_login_device`;
ALTER TABLE sys_users ADD COLUMN `lock_until`         DATETIME     DEFAULT NULL COMMENT '锁定截止时间' AFTER `login_fail_count`;
ALTER TABLE sys_users ADD COLUMN `password_changed_at` DATETIME    DEFAULT NULL COMMENT '密码修改时间' AFTER `lock_until`;

-- 账号注销
ALTER TABLE sys_users ADD COLUMN `logout_at`  DATETIME DEFAULT NULL COMMENT '注销时间' AFTER `password_changed_at`;

-- 新增索引
ALTER TABLE sys_users ADD INDEX `idx_email` (`email`);
ALTER TABLE sys_users ADD INDEX `idx_phone` (`phone`);
```

## 二、sys_login_devices — 登录设备表

管理用户的多端登录状态，支持设备信任、踢出设备等功能。

```sql
CREATE TABLE IF NOT EXISTS `sys_login_devices` (
    `id`              BIGINT UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    `user_id`         BIGINT UNSIGNED NOT NULL COMMENT '用户ID',
    `device_id`       VARCHAR(100) NOT NULL COMMENT '设备唯一标识',
    `device_type`     VARCHAR(20)  NOT NULL COMMENT '设备类型: web/h5/app/miniapp',
    `device_name`     VARCHAR(100) DEFAULT '' COMMENT '设备名称',
    `browser`         VARCHAR(50)  DEFAULT '' COMMENT '浏览器',
    `os`              VARCHAR(50)  DEFAULT '' COMMENT '操作系统',
    `ip`              VARCHAR(50)  DEFAULT '' COMMENT 'IP地址',
    `location`        VARCHAR(100) DEFAULT '' COMMENT '登录地点',
    `app_version`     VARCHAR(50)  DEFAULT '' COMMENT 'App版本',
    `system_version`  VARCHAR(100) DEFAULT '' COMMENT '系统版本',
    `device_model`    VARCHAR(100) DEFAULT '' COMMENT '设备型号',
    `platform`        VARCHAR(50)  DEFAULT '' COMMENT '平台: web/h5/ios/android/miniapp/windows/macos/linux',
    `channel`         VARCHAR(50)  DEFAULT '' COMMENT '渠道',
    `token_jti`       VARCHAR(100) NOT NULL COMMENT 'JWT ID',
    `last_active_at`  DATETIME     DEFAULT NULL COMMENT '最后活跃时间',
    `is_current`      TINYINT      DEFAULT 0  COMMENT '是否当前设备',
    `created_at`      DATETIME(3)  NOT NULL DEFAULT CURRENT_TIMESTAMP(3) COMMENT '创建时间',
    `updated_at`      DATETIME(3)  NOT NULL DEFAULT CURRENT_TIMESTAMP(3) ON UPDATE CURRENT_TIMESTAMP(3) COMMENT '更新时间',
    PRIMARY KEY (`id`),
    KEY `idx_user_id` (`user_id`),
    KEY `idx_device_id` (`device_id`),
    KEY `idx_token_jti` (`token_jti`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='登录设备表';
```

**使用场景**：
- 登录时记录设备信息
- 多端登录管理（查看/踢出）
- 设备信任（常用设备免验证）
- 异地登录告警

## 三、sys_security_logs — 安全日志表

记录所有安全相关操作，用于审计和异常检测。

```sql
CREATE TABLE IF NOT EXISTS `sys_security_logs` (
    `id`           BIGINT UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    `user_id`      BIGINT UNSIGNED NOT NULL COMMENT '用户ID',
    `event_type`   VARCHAR(30)  NOT NULL COMMENT '事件类型',
    `event_detail` VARCHAR(500) DEFAULT '' COMMENT '事件详情',
    `ip`           VARCHAR(50)  DEFAULT '' COMMENT 'IP地址',
    `user_agent`   VARCHAR(500) DEFAULT '' COMMENT 'User-Agent',
    `status`       TINYINT      DEFAULT 1  COMMENT '状态: 0失败 1成功',
    `created_at`   DATETIME(3)  NOT NULL DEFAULT CURRENT_TIMESTAMP(3) COMMENT '创建时间',
    PRIMARY KEY (`id`),
    KEY `idx_user_id` (`user_id`),
    KEY `idx_event_type` (`event_type`),
    KEY `idx_created_at` (`created_at`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='安全日志表';
```

**事件类型枚举**：

| event_type | 说明 |
|-----------|------|
| login | 登录 |
| logout | 登出 |
| login_fail | 登录失败 |
| password_change | 修改密码 |
| password_reset | 重置密码 |
| bind_oauth | 绑定第三方 |
| unbind_oauth | 解绑第三方 |
| account_lock | 账号锁定 |
| account_deactivate | 账号注销 |

## 四、sys_oauth_users — 第三方登录绑定表

支持微信、GitHub、Google 等第三方账号登录。

```sql
CREATE TABLE IF NOT EXISTS `sys_oauth_users` (
    `id`                BIGINT UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    `user_id`           BIGINT UNSIGNED NOT NULL COMMENT '用户ID',
    `provider`          VARCHAR(20)  NOT NULL COMMENT '提供商: wechat/github/google',
    `provider_type`     VARCHAR(20)  DEFAULT '' COMMENT '提供商类型: oauth/oauth2/oidc',
    `provider_user_id`  VARCHAR(100) NOT NULL COMMENT '第三方用户ID',
    `provider_username` VARCHAR(100) DEFAULT '' COMMENT '第三方用户名',
    `provider_avatar`   VARCHAR(255) DEFAULT '' COMMENT '第三方头像',
    `access_token`      VARCHAR(500) DEFAULT '' COMMENT 'Access Token',
    `refresh_token`     VARCHAR(500) DEFAULT '' COMMENT 'Refresh Token',
    `expires_at`        DATETIME     DEFAULT NULL COMMENT 'Token过期时间',
    `created_at`        DATETIME(3)  NOT NULL DEFAULT CURRENT_TIMESTAMP(3) COMMENT '创建时间',
    `updated_at`        DATETIME(3)  NOT NULL DEFAULT CURRENT_TIMESTAMP(3) ON UPDATE CURRENT_TIMESTAMP(3) COMMENT '更新时间',
    PRIMARY KEY (`id`),
    KEY `idx_user_id` (`user_id`),
    UNIQUE KEY `uk_provider_user` (`provider`, `provider_user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='第三方登录绑定表';
```

**使用场景**：
- 微信小程序/公众号登录
- GitHub/Google OAuth 登录
- 账号绑定/解绑管理

## 六、sys_password_history — 密码历史表

防止用户重复使用旧密码。

```sql
CREATE TABLE IF NOT EXISTS `sys_password_history` (
    `id`         BIGINT UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    `user_id`    BIGINT UNSIGNED NOT NULL COMMENT '用户ID',
    `password`   VARCHAR(255) NOT NULL COMMENT '历史密码(bcrypt)',
    `created_at` DATETIME(3)  NOT NULL DEFAULT CURRENT_TIMESTAMP(3) COMMENT '创建时间',
    PRIMARY KEY (`id`),
    KEY `idx_user_id` (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='密码历史表';
```

**使用规则**：
- 每次修改密码时，将旧密码存入此表
- 修改密码时检查最近 N 次（建议 5 次）是否重复
- 定期清理超过 6 个月的历史记录

## 七、sys_user_real_names — 实名认证表

```sql
CREATE TABLE IF NOT EXISTS `sys_user_real_names` (
    `id`            BIGINT UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '主键ID',

    `updated_at`    DATETIME(3)  NOT NULL DEFAULT CURRENT_TIMESTAMP(3) ON UPDATE CURRENT_TIMESTAMP(3) COMMENT '更新时间',
    `deleted_at`    DATETIME(3)  DEFAULT NULL COMMENT '删除时间',

    UNIQUE KEY `uk_user_id` (`user_id`),
```
